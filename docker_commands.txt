# После написания Dockerfile для создания образа необходимо, 
# находясь в директории с файлом Dockerfile, в терминале запустить команду:
docker build .
ИЛИ
docker build -t pupupu .

# Посмотреть существующие образы
docker images

# После построения образа в терминале будет выведен хэш — название контейнера.
# Эта команда запустит образ 2f00557c8d5e и создаст контейнер с приложением. 
# Под приложением здесь имеется в виду наш API, но в действительности может быть любым приложением (базой, Redis, фронтенд и т.д.).
# Для запуска образа в контейнере необходимо прописать команду:
docker run -p 8888:8000 pupupu (левый порт это порт, по которому мы получает доступ к докеру; правый порт это порт, на котором запускается само приложение)

# Скачивание образа из интернета (докер хаба)
docker pull redis:7.4
