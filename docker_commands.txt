# После написания Dockerfile для создания образа необходимо, 
# находясь в директории с файлом Dockerfile, в терминале запустить команду:
docker build .
ИЛИ
docker build -t pupupu .

# Посмотреть существующие образы
docker images

# После построения образа в терминале будет выведен хэш — название контейнера.
# Эта команда запустит образ 2f00557c8d5e и создаст контейнер с приложением. 
# Под приложением здесь имеется в виду наш API, но в действительности может быть любым приложением (базой, Redis, фронтенд и т.д.).
# Для запуска образа в контейнере необходимо прописать команду:
docker run -p 8888:8000 pupupu (левый порт это порт, по которому мы получает доступ к докеру; правый порт это порт, на котором запускается само приложение)

# Скачивание образа из интернета (докер хаба)
# В дальнейшем его можно запускать сразу
docker pull redis:7.4

### Основные команды стандартного докера

# Удалить все остановленные контейнеры, неиспользуемые сети, "dangling" образы (без имени/тега) и кэш сборок. Безопасная базовая очистка.
docker system prune

# То же самое, но удалить ВСЕ образы, не используемые запущенными контейнерами (а не только dangling). Освобождает очень много места.
docker system prune -a

# Глобальная зачистка "под ноль": удалит всё вышеперечисленное + все неиспользуемые volumes. ОСТОРОЖНО: удалит данные всех остановленных БД!
docker system prune -a --volumes

# Удалить только остановленные контейнеры
docker container prune

# Удалить только "dangling" образы (остатки после пересборки, у которых вместо имени стоит <none>)
docker image prune

# Удалить ВСЕ образы, которые не используются ни одним запущенным контейнером
docker image prune -a

# Удалить все volumes, которые не привязаны ни к одному контейнеру. ОСТОРОЖНО: удалит данные БД, если контейнеры удалены/остановлены.
docker volume prune

# Удалить все сети, которые не используются ни одним контейнером
docker network prune

# Очистить кэш сборок Docker BuildKit (часто занимает гигабайты на диске)
docker builder prune


# Показать только запущенные в данный момент контейнеры
docker ps

# Показать абсолютно все контейнеры (включая остановленные и упавшие с ошибками)
docker ps -a

# Показать все локальные Docker-образы и их размер
docker images

# Показать все созданные volumes (хранилища данных)
docker volume ls

# Показать все созданные сети
docker network ls

# Остановить работающий контейнер
docker stop <имя_или_id_контейнера>

# Удалить контейнер (должен быть предварительно остановлен)
docker rm <имя_или_id_контейнера>

# Принудительно остановить и сразу удалить контейнер
docker rm -f <имя_или_id_контейнера>

# Удалить Docker-образ
docker rmi <имя_или_id_образа>

# Удалить volume (хранилище с данными)
docker volume rm <имя_volume>

# Удалить сеть
docker network rm <имя_сети>

# Посмотреть логи контейнера (весь накопленный вывод)
docker logs <имя_или_id_контейнера>

# Посмотреть логи контейнера в реальном времени (как tail -f)
docker logs -f <имя_или_id_контейнера>

# Зайти внутрь работающего контейнера через интерактивный терминал (bash или sh)
docker exec -it <имя_или_id_контейнера> bash

# Посмотреть детальную техническую информацию о контейнере (IP-адрес, настройки сети, переменные окружения)
docker inspect <имя_или_id_контейнера>

# Показать статистику использования ресурсов (CPU, RAM) контейнерами в реальном времени
docker stats


### docker compose

# Поднять все связанные контейнеры в фоновом режиме, запустить сборку образа, сети и тд
# Нужно находиться в директории где присутствует docker-compose.yml
docker compose up --build -d

# --force-recreate гарантирует, что контейнеры пересоздадутся даже если Docker решит, что они "не изменились".
docker compose up -d --build --force-recreate

# Логи конкретного контейнера
docker compose logs -f booking_back_service

# Зайти внутрь работающего контейнера (как ssh)
docker compose exec booking_back_service bash

# Перезапустить только один сервис (не трогая остальные)
docker compose restart booking_celery_worker_service

# Останавливает контейнеры, но не удаляет их
# Контейнеры остаются в системе, сеть остается, volumes с данными остаются.
docker compose stop

# После этого можно без пересборки поднять контейнеры
docker compose start
ИЛИ
docker compose up -d

# Аккуратно разобрать контейнеры
# Останавливает контейнеры и удаляет их, а также удаляет сеть. 
# Но volumes (данные БД) остаются нетронутыми.
docker compose down

# То же самое, что down, но удаляет еще и volumes. 
# Это значит, что вся база данных будет удалена.
docker compose down -v

# Удаляет контейнеры, сеть и собранные Docker-образы вашего проекта.
docker compose down --rmi all

# Удаляет всё: контейнеры, сеть, volumes (данные БД) и образы.
docker compose down -v --rmi all
