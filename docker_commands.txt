# После написания Dockerfile для создания образа необходимо, 
# находясь в директории с файлом Dockerfile, в терминале запустить команду:
docker build .
ИЛИ
docker build -t pupupu .

# Посмотреть существующие образы
docker images

# После построения образа в терминале будет выведен хэш — название контейнера.
# Эта команда запустит образ 2f00557c8d5e и создаст контейнер с приложением. 
# Под приложением здесь имеется в виду наш API, но в действительности может быть любым приложением (базой, Redis, фронтенд и т.д.).
# Для запуска образа в контейнере необходимо прописать команду:
docker run -p 8888:8000 pupupu (левый порт это порт, по которому мы получает доступ к докеру; правый порт это порт, на котором запускается само приложение)

# Скачивание образа из интернета (докер хаба)
# В дальнейшем его можно запускать сразу
docker pull redis:7.4

# Остановить контейнер
docker stop pupupu

# Удалить образ
docker rmi 5edf6kh
docker rmi -f 5edf6kh (принудительно)

# Удалить контейнер
docker rm pupupu


### docker compose

# Поднять все связанные контейнеры в фоновом режиме, запустить сборку образа, сети и тд
# Нужно находиться в директории где присутствует docker-compose.yml
docker compose up --build -d

# --force-recreate гарантирует, что контейнеры пересоздадутся даже если Docker решит, что они "не изменились".
docker compose up -d --build --force-recreate

# Логи конкретного контейнера
docker compose logs -f booking_back_service

# Зайти внутрь работающего контейнера (как ssh)
docker compose exec booking_back_service bash

# Перезапустить только один сервис (не трогая остальные)
docker compose restart booking_celery_worker_service

# Останавливает контейнеры, но не удаляет их
# Контейнеры остаются в системе, сеть остается, volumes с данными остаются.
docker compose stop

# После этого можно без пересборки поднять контейнеры
docker compose start
ИЛИ
docker compose up -d

# Аккуратно разобрать контейнеры
# Останавливает контейнеры и удаляет их, а также удаляет сеть. 
# Но volumes (данные БД) остаются нетронутыми.
docker compose down

# То же самое, что down, но удаляет еще и volumes. 
# Это значит, что вся база данных будет удалена.
docker compose down -v

# Удаляет контейнеры, сеть и собранные Docker-образы вашего проекта.
docker compose down --rmi all

# Удаляет всё: контейнеры, сеть, volumes (данные БД) и образы.
docker compose down -v --rmi all
